# backend/app/services/ingestion/extractors/csv_extractor.py
import csv


def extract_from_csv(file_path: str) -> str:
    rows = []
    with open(file_path, newline="", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        headers = reader.fieldnames or []
        rows.append(", ".join(headers))
        for row in reader:
            rows.append(", ".join(f"{k}: {v}" for k, v in row.items() if v))
    return "\n".join(rows)
```

# backend/app/services/ingestion/extractors/sitemap_extractor.py
import xml.etree.ElementTree as ET

import httpx


def extract_urls_from_sitemap(sitemap_url: str, max_urls: int = 200) -> list[str]:
    """Fetch sitemap XML and return all <loc> URLs found."""
    response = httpx.get(sitemap_url, follow_redirects=True, timeout=30)
    response.raise_for_status()

    root = ET.fromstring(response.text)

    # Strip XML namespaces for simple matching
    ns = {"sm": "http://www.sitemaps.org/schemas/sitemap/0.9"}
    urls: list[str] = []

    # Handle sitemap index (list of sitemaps)
    for sitemap_tag in root.findall("sm:sitemap", ns):
        loc = sitemap_tag.find("sm:loc", ns)
        if loc is not None and loc.text:
            urls.extend(extract_urls_from_sitemap(loc.text.strip(), max_urls - len(urls)))
            if len(urls) >= max_urls:
                break

    # Handle regular urlset
    for url_tag in root.findall("sm:url", ns):
        loc = url_tag.find("sm:loc", ns)
        if loc is not None and loc.text:
            urls.append(loc.text.strip())
            if len(urls) >= max_urls:
                break

    return urls[:max_urls]
```

```python
# backend/app/workers/tasks/auto_recharge.py
import asyncio
import logging
import uuid

from sqlalchemy import select

from app.database import async_session_factory
from app.models.organizational import Workspace
from app.services import credits as credits_service
from app.workers.celery_app import celery_app

logger = logging.getLogger(__name__)

CREDITS_PRICE_ID = "price_1000_credits"  # Placeholder — set real Stripe price ID via env


@celery_app.task
def trigger_auto_recharge(workspace_id: str) -> dict:
    return asyncio.run(_recharge(uuid.UUID(workspace_id)))


async def _recharge(workspace_id: uuid.UUID) -> dict:
    async with async_session_factory() as session:
        try:
            result = await session.execute(
                select(Workspace).where(Workspace.id == workspace_id)
            )
            workspace = result.scalar_one_or_none()
            if not workspace:
                return {"status": "error", "detail": "Workspace not found"}

            if not workspace.auto_recharge_enabled:
                return {"status": "skipped", "detail": "Auto-recharge disabled"}

            if not workspace.stripe_customer_id:
                return {"status": "error", "detail": "No Stripe customer"}

            amount = workspace.auto_recharge_amount

            try:
                import stripe
                from app.config import settings
                stripe.api_key = settings.STRIPE_SECRET_KEY

                # Create invoice item then invoice and pay immediately
                stripe.InvoiceItem.create(
                    customer=workspace.stripe_customer_id,
                    amount=int(amount * 0.04 * 100),  # $0.04 per credit, in cents
                    currency="usd",
                    description=f"Pulse credit top-up ({amount} credits)",
                )
                invoice = stripe.Invoice.create(
                    customer=workspace.stripe_customer_id,
                    auto_advance=True,
                )
                invoice = stripe.Invoice.finalize_invoice(invoice.id)
                stripe.Invoice.pay(invoice.id)

            except Exception as e:
                logger.error(f"Stripe auto-recharge failed for {workspace_id}: {e}")
                return {"status": "error", "detail": str(e)}

            # Credit the workspace
            new_balance = await credits_service.add_credits(
                session, workspace_id, amount,
                reason="auto_recharge",
            )
            await session.commit()

            logger.info(f"Auto-recharged {amount} credits for workspace {workspace_id}. New balance: {new_balance}")
            return {"status": "success", "credits_added": amount, "new_balance": new_balance}

        except Exception as e:
            await session.rollback()
            logger.error(f"Auto-recharge error: {e}")
            return {"status": "error", "detail": str(e)}
```

### Step 7d: Hook into debit_credits

**Step 6: Trigger recharge after debit falls below threshold**

In `backend/app/services/credits.py`, in the `debit_credits` function, after the ledger entry:

```python
async def debit_credits(
    db: AsyncSession,
    workspace_id: uuid.UUID,
    amount: int,
    reason: str,
    reference_id: str | None = None,
) -> int:
    result = await db.execute(
        update(Workspace)
        .where(Workspace.id == workspace_id, Workspace.credit_balance >= amount)
        .values(credit_balance=Workspace.credit_balance - amount)
        .returning(Workspace.credit_balance, Workspace.auto_recharge_enabled,
                   Workspace.auto_recharge_threshold)
    )
    row = result.one_or_none()
    if row is None:
        raise ValueError("Insufficient credit balance")

    new_balance, auto_recharge_enabled, threshold = row[0], row[1], row[2]

    ledger_entry = CreditLedger(
        workspace_id=workspace_id,
        amount=-amount,
        reason=reason,
        reference_id=reference_id,
        balance_after=new_balance,
    )
    db.add(ledger_entry)
    await db.flush()

    # Trigger auto-recharge if balance dropped below threshold
    if auto_recharge_enabled and new_balance <= threshold:
        from app.workers.tasks.auto_recharge import trigger_auto_recharge
        trigger_auto_recharge.delay(str(workspace_id))

    return new_balance
```
